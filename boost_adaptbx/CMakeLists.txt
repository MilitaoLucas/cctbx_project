# boost_adaptbx module

# Generate type_id_eq.h header file
# This detects which unsigned type matches size_t
set(TYPE_ID_TEST_CODE "
#include <typeinfo>
#include <iostream>
#include <stddef.h>

int main() {
  std::cout
    << static_cast<int>(typeid(unsigned short) == typeid(size_t))
    << static_cast<int>(typeid(unsigned) == typeid(size_t))
    << static_cast<int>(typeid(unsigned long) == typeid(size_t))
    << static_cast<int>(typeid(unsigned long long) == typeid(size_t))
    << std::endl;
  return 0;
}
")

file(WRITE "${CMAKE_BINARY_DIR}/type_id_test.cpp" "${TYPE_ID_TEST_CODE}")

try_run(
    TYPE_ID_RUN_RESULT
    TYPE_ID_COMPILE_RESULT
    "${CMAKE_BINARY_DIR}"
    "${CMAKE_BINARY_DIR}/type_id_test.cpp"
    RUN_OUTPUT_VARIABLE TYPE_ID_OUTPUT
)

if(NOT TYPE_ID_COMPILE_RESULT OR NOT TYPE_ID_RUN_RESULT EQUAL 0)
    message(FATAL_ERROR "Failed to detect size_t type mapping")
endif()

string(STRIP "${TYPE_ID_OUTPUT}" TYPE_ID_OUTPUT)

# Generate the header file
set(TYPE_ID_EQ_H_PATH "${CMAKE_BINARY_DIR}/include/boost_adaptbx/type_id_eq.h")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/include/boost_adaptbx")

set(TYPE_ID_DEFINES "")
set(UNSIGNED_TYPES "unsigned short" "unsigned" "unsigned long" "unsigned long long")
set(UNSIGNED_DEFINES "UNSIGNED_SHORT" "UNSIGNED" "UNSIGNED_LONG" "UNSIGNED_LONG_LONG")
list(LENGTH UNSIGNED_TYPES NUM_TYPES)
math(EXPR MAX_INDEX "${NUM_TYPES} - 1")

foreach(INDEX RANGE ${MAX_INDEX})
    string(SUBSTRING "${TYPE_ID_OUTPUT}" ${INDEX} 1 CHAR)
    if(CHAR STREQUAL "1")
        list(GET UNSIGNED_DEFINES ${INDEX} DEFINE_NAME)
        set(TYPE_ID_DEFINES "${TYPE_ID_DEFINES}#define BOOST_ADAPTBX_TYPE_ID_SIZE_T_EQ_${DEFINE_NAME}\n")
    endif()
endforeach()

file(WRITE "${TYPE_ID_EQ_H_PATH}"
"// This file is auto-generated by CMake during configuration
// Based on: cctbx_project/boost_adaptbx/SConscript

#ifndef BOOST_ADAPTBX_TYPE_ID_EQ_H
#define BOOST_ADAPTBX_TYPE_ID_EQ_H

${TYPE_ID_DEFINES}
#endif // BOOST_ADAPTBX_TYPE_ID_EQ_H
")

message(STATUS "Generated boost_adaptbx/type_id_eq.h")
